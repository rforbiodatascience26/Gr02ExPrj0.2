# Calling the first row of the two expression matrices, as tibbles, which are both transformed to extract their linearized values as pure vectors, which are converted to pcw, unless they're already born whereupon their unit is converted to years (yrs).

ltnc <- "(?<=plus0\\.7_)-?\\d+\\.\\d+"
hpc_colnames <- tibble(sample=names(hippocampus_expression_matrix))|> mutate(age_log = as.numeric(str_extract(sample, ltnc))) |> filter(!is.na(age_log)) |> mutate(age = round(((10^age_log-0.7)*52+40),1)) |> mutate(age = if_else(age >= 40, str_c(round((age-40)/52,1),"yrs"),str_c(age,"pcw"))) 

dlpfc_colnames <- tibble(sample=names(DLPFC_expression_matrix))|> mutate(age_log = as.numeric(str_extract(sample, ltnc))) |> filter(!is.na(age_log)) |> mutate(age = round(((10^age_log-0.7)*52+40),1)) |> mutate(age = if_else(age >= 40, str_c(round((age-40)/52,1),"yrs"),str_c(age,"pcw")))
